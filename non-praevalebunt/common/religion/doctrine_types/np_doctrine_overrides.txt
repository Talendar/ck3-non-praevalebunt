# Non Praevalebunt - doctrine overrides (works with or without Shadows of the Cross)
# A full copy of the vanilla definition (a definition can only be replaced whole), with the Non Praevalebunt
# change marked by a comment. This file must sort after 20_doctrines.txt (and SOTC's ac_doctrines.txt).

doctrine_pluralism_righteous = {
	doctrine_group_type = doctrine_pluralism
	index = 1
	divergence = 5

	icon = "doctrine_pluralism_righteous"

	piety_cost = {
		value = faith_doctrine_cost_mid
		if = {
			limit = { rite_has_doctrine = doctrine_pluralism_righteous }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}
	can_pick = {
		# Non Praevalebunt: within the Church, Righteous may stand with Adaptability. This mirrors the exception
		# in tenet_adaptive (np_tenet_overrides.txt), so the rite editor accepts e.g. the Mozarabic rite.
		custom_description = {
			text = np_righteous_adaptive_trigger
			OR = {
				NOT = { tenet:tenet_adaptive = { is_in_list = selected_doctrines } }
				AND = {
					exists = rite:roman_rite
					this = rite:roman_rite.faith
				}
			}
		}
	}

	parameters = {
		pluralism_righteous_revocation_tyranny_reduced
	}

	special_parameters = {
		hostility_towards_us_mult = 1
	}

	character_modifier = {
		minority_different_faith_opinion = -10
	}
}

# The two clerical marriage doctrines: vanilla copies plus the rite window lock (v0.2).

doctrine_clerical_marriage_allowed = {
	doctrine_group_type = doctrine_clerical_marriage
	index = 0
	divergence = 0

	# Non Praevalebunt: the rite window lock. Every Christian rite holds one of these two doctrines, and the game
	# re-checks the doctrines a rite holds before it lets an edit through, so this blocks "Confirm" while the
	# selection holds anything the Church forbids (np_selection_is_clean_trigger). Shadows of the Cross locks its
	# own rules the same way, through its origin doctrines. Neither doctrine had a can_pick before.
	can_pick = {
		custom_description = {
			text = np_rite_window_lock_trigger
			np_selection_is_clean_trigger = yes
		}
	}

	piety_cost = {
		value = faith_doctrine_cost_low
		if = {
			limit = { rite_has_doctrine = doctrine_clerical_marriage_allowed }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_clerical_marriage_disallowed }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
	}

	# not defining the parameter means it is false by default
	parameters = {
		clergy_can_marry
	}
}

doctrine_clerical_marriage_disallowed = {
	doctrine_group_type = doctrine_clerical_marriage
	index = 1
	divergence = 15

	# Non Praevalebunt: the rite window lock. Every Christian rite holds one of these two doctrines, and the game
	# re-checks the doctrines a rite holds before it lets an edit through, so this blocks "Confirm" while the
	# selection holds anything the Church forbids (np_selection_is_clean_trigger). Shadows of the Cross locks its

	can_pick = {

			text = np_rite_window_lock_trigger

	}

	parameters = {

	}

	piety_cost = {
		value = faith_doctrine_cost_low
		if = {
			limit = { rite_has_doctrine = doctrine_clerical_marriage_disallowed }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_clerical_marriage_allowed }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
	}
}
