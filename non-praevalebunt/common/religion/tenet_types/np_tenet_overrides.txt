tenet_adaptive = {
	icon = tenet_adaptive

	piety_cost = {
		if = {
			limit = {
				OR = {
					rite_religion_tag = christianity_religion
					rite_religion_tag = islam_religion
				}
			}
			value = faith_tenet_cost_high
		}
		else_if = {
			limit = {
				OR = {
					rite_religion_tag = judaism_religion
					rite_religion_tag = buddhism_religion
					rite_religion_tag = jainism_religion
					rite_has_doctrine = pagan_hostility_doctrine
				}
			}
			value = faith_tenet_cost_low
		}
		else = {
			value = faith_tenet_cost_mid
		}

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_adaptive }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	can_pick = {
		# Non Praevalebunt: the Church forbids Pluralist (it denies that one teaching is true), so within the
		# Church Adaptability is also allowed with Righteous. Fundamentalist still excludes it.
		custom_description = {
			text = np_adaptive_pluralism_trigger
			OR = {
				NOR = {
					doctrine:doctrine_pluralism_fundamentalist = { is_in_list = selected_doctrines }
					doctrine:doctrine_pluralism_righteous = { is_in_list = selected_doctrines }
				}
				AND = {
					exists = rite:roman_rite
					this = rite:roman_rite.faith
					doctrine:doctrine_pluralism_righteous = { is_in_list = selected_doctrines }
				}
			}
		}
	}

	parameters = {
		tenet_adaptive_conversion_resistance
		reduced_vassal_religion_discontent
		allow_adopt_any_holy_site
		tenet_adaptive_study_rite_bonus
	}

	character_modifier = {
		different_faith_popular_opinion_mult = -0.25
		different_faith_liege_opinion = 15
		different_faith_suzerain_opinion = 15
		opinion_of_different_faith_liege = 15
		opinion_of_different_faith_suzerain = 15
		trade_route_length_add = 1
		trade_competitiveness_add = 2
	}

	personal_tenet_modifier = {
		minority_opinion = 10
		minority_tax_contribution_mult = 0.1
		monthly_diplomacy_lifestyle_xp_gain_mult = 0.2
		study_faith_scheme_phase_duration_add = medium_scheme_phase_duration_bonus_value
		learn_language_scheme_phase_duration_add = medium_scheme_phase_duration_bonus_value
		max_learn_language_schemes_add = 1
		trade_competitiveness_mult = 0.05
	}

	personal_tenet_parameters = {
		adaptive_study_rite_cultural_acceptance
		tenet_adaptive_study_rite_bonus
	}

	traits = {
		virtues = {
			# Non Praevalebunt: humble instead of fickle. Inconstancy is a vice, not a virtue (James 1:8).
			humble = { weight = 20 scale = 1 }
		}
		sins = {
			stubborn = { weight = 20 scale = 1 }
		}
	}
}
