# Rite scope. If this rite is in good standing and holds anything forbidden, set it right.
# No message of our own: the game already sends its "Rite updated" message for every change, listing
# exactly what changed. A one-time notice at game start explains why such changes happen.
np_guard_rite_effect = {
	# A "dangerous rite" heresy outbreak (kind 2) creates a rite 35-50 points from Rome, below the heresy
	# line. It is born fallen (Tier 3), so mark it before it could be mistaken for a rite in good standing.
	# Traces left by the game: the spread budget on the rite, or a heresiarch founder leading it.
	if = {
		limit = {
			np_rite_in_church_trigger = yes
			np_rite_is_tier1_trigger = no
			NOT = { has_variable = np_good_standing }
			NOT = { has_variable = np_fallen }
			OR = {
				has_variable = c2_spread_remaining_converts
				head_of_rite ?= { has_trait = heresiarch }
				rite_founder ?= { has_trait = heresiarch }
			}
		}
		set_variable = np_fallen
		debug_log = "Non Praevalebunt: a heresy outbreak rite was marked as fallen (Tier 3)"
		debug_log_scopes = yes
	}
	# Remember that this rite was seen in good standing (see np_rite_is_heretic_trigger).
	if = {
		limit = {
			np_rite_is_guarded_trigger = yes
			NOT = { has_variable = np_good_standing }
		}
		set_variable = np_good_standing
	}
	if = {
		limit = {
			np_rite_is_guarded_trigger = yes
			OR = {
				np_rite_has_forbidden_doctrine_trigger = yes
				np_rite_has_forbidden_tenet_trigger = yes
			}
		}
		save_scope_as = np_fix_rite
		np_fix_doctrines_effect = yes
		np_fix_tenets_effect = yes
		# Again: a replacement tenet (e.g. Be Fruitful) can make a doctrine forbidden in combination.
		np_fix_doctrines_effect = yes
		debug_log = "Non Praevalebunt: guard corrected a rite"
		debug_log_scopes = yes
	}
}

# Rite scope. Swap each forbidden doctrine for the nearest allowed one in its group.
np_fix_doctrines_effect = {
	# Faith and Church
	if = { limit = { rite_has_doctrine = doctrine_polytheist } change_rite_doctrine = doctrine_monotheist }
	if = {
		limit = {
			this = faith.main_rite
			OR = {
				rite_has_doctrine = doctrine_no_head
				rite_has_doctrine = doctrine_temporal_head
			}
		}
		change_rite_doctrine = doctrine_spiritual_head
	}
	if = { limit = { rite_has_doctrine = doctrine_pluralism_pluralistic } change_rite_doctrine = doctrine_pluralism_righteous }
	if = { limit = { rite_has_doctrine = doctrine_theocracy_lay_clergy } change_rite_doctrine = doctrine_theocracy_temporal }
	if = { limit = { rite_has_doctrine = doctrine_sacraments_absent } change_rite_doctrine = doctrine_sacraments_central }
	if = {
		limit = {
			OR = {
				rite_has_doctrine = doctrine_clerical_gender_either
				rite_has_doctrine = doctrine_clerical_gender_female_only
			}
		}
		change_rite_doctrine = doctrine_clerical_gender_male_only
	}
	# Marriage and family
	if = {
		limit = {
			OR = {
				rite_has_doctrine = doctrine_polygamy
				rite_has_doctrine = doctrine_concubines
			}
		}
		change_rite_doctrine = doctrine_monogamy
	}
	if = { limit = { rite_has_doctrine = doctrine_divorce_allowed } change_rite_doctrine = doctrine_divorce_approval }
	if = {
		limit = {
			rite_has_doctrine = doctrine_divorce_disallowed
			rite_has_tenet = tenet_be_fruitful_and_multiply
		}
		change_rite_doctrine = doctrine_divorce_approval

				rite_has_doctrine = doctrine_consanguinity_aunt_nephew_and_uncle_niece
				rite_has_doctrine = doctrine_consanguinity_unrestricted
			}
		}
		change_rite_doctrine = doctrine_consanguinity_cousins
	}
	# Sins and crimes
	if = { limit = { rite_has_doctrine = doctrine_homosexuality_accepted } change_rite_doctrine = doctrine_homosexuality_shunned }
	if = {
		limit = {
			OR = {
				rite_has_doctrine = doctrine_deviancy_accepted
				rite_has_doctrine = doctrine_deviancy_virtuous
			}
		}
		change_rite_doctrine = doctrine_deviancy_shunned
	}
	if = { limit = { rite_has_doctrine = doctrine_adultery_men_accepted } change_rite_doctrine = doctrine_adultery_men_shunned }
	if = { limit = { rite_has_doctrine = doctrine_adultery_women_accepted } change_rite_doctrine = doctrine_adultery_women_shunned }
	if = { limit = { rite_has_doctrine = doctrine_kinslaying_accepted } change_rite_doctrine = doctrine_kinslaying_shunned }
	if = {
		limit = {
			OR = {
				rite_has_doctrine = doctrine_witchcraft_accepted
				rite_has_doctrine = doctrine_witchcraft_virtuous
			}
		}
		change_rite_doctrine = doctrine_witchcraft_shunned
	}
	if = {
		limit = {
			OR = {
				rite_has_doctrine = doctrine_usury_discouraged
				rite_has_doctrine = doctrine_usury_outsiders_only
				rite_has_doctrine = doctrine_usury_accepted
			}
		}
		change_rite_doctrine = doctrine_usury_forbidden
	}
	# Clergy and practice
	if = {
		limit = {
			OR = {
				rite_has_doctrine = doctrine_monasticism_absent
				rite_has_doctrine = doctrine_monasticism_forbidden
			}
		}
		change_rite_doctrine = doctrine_monasticism_accepted
	}
	if = { limit = { rite_has_doctrine = doctrine_pilgrimage_forbidden } change_rite_doctrine = doctrine_pilgrimage_encouraged }

	if = {
		limit = {
			OR = {

				rite_has_doctrine = doctrine_funeral_cremation
				rite_has_doctrine = doctrine_funeral_sky_burial
				rite_has_doctrine = doctrine_funeral_mummification
			}
		}
		change_rite_doctrine = doctrine_funeral_stoic
	}
	if = {
		limit = {
			OR = {
				rite_has_doctrine = special_doctrine_meat_only
				rite_has_doctrine = special_doctrine_meat_forbidden_fish
				rite_has_doctrine = special_doctrine_meat_forbidden
				rite_has_doctrine = special_doctrine_nothing_of_the_slain
			}
		}
		change_rite_doctrine = doctrine_diet_unrestricted
	}
	if = { limit = { rite_has_doctrine = special_doctrine_alcohol_forbidden } change_rite_doctrine = doctrine_alcohol_permitted }

# Rite scope, saved as scope:np_fix_rite. Remove each forbidden core tenet and put an allowed one in its place.
# Vanilla caps core tenets at 3, Shadows of the Cross at 5, so five passes always cover them all.
# Extra passes do nothing. (No list is used, so nothing leaks from one rite to the next during a sweep.)
np_fix_tenets_effect = {
	np_fix_one_tenet_effect = yes
	np_fix_one_tenet_effect = yes
	np_fix_one_tenet_effect = yes
	np_fix_one_tenet_effect = yes
	np_fix_one_tenet_effect = yes
}

# Rite scope, saved as scope:np_fix_rite.
# No "status" filter on the first iterator: it must look at the rite's own core tenets (see np_triggers.txt).
np_fix_one_tenet_effect = {
	if = {
		limit = { np_rite_has_forbidden_tenet_trigger = yes }
		clear_saved_scope = np_bad_tenet
		random_rite_tenet = {
			limit = { np_tenet_is_forbidden_trigger = yes }
			save_scope_as = np_bad_tenet
		}
		if = {
			limit = { exists = scope:np_bad_tenet }
			remove_rite_tenet = scope:np_bad_tenet
			np_add_replacement_tenet_effect = yes
			np_after_tenet_swap_effect = yes
		}
	}
}

# Rite scope (scope:np_fix_rite). scope:np_bad_tenet was just removed and scope:np_new_tenet (if it
# exists) was added.
# Shadows of the Cross compatibility, automatic: SOTC records each rite's two "canon" tenets in the rite
# variables ac_foundation_1 / ac_foundation_2 and puts a removed canon tenet back. Point the record at the
# replacement so SOTC keeps the swap; with no replacement, clear it so SOTC refills it from the rite's
# current (allowed) tenets. Without SOTC these variables never exist, so this does nothing and names no
# SOTC key. Each slot is checked with has_variable first: reading an unset variable logs an error.
np_after_tenet_swap_effect = {
	if = {
		limit = {
			has_variable = ac_foundation_1
			var:ac_foundation_1 = scope:np_bad_tenet
		}
		if = {
			limit = { exists = scope:np_new_tenet }
			set_variable = { name = ac_foundation_1 value = scope:np_new_tenet }
		}
		else = {
			remove_variable = ac_foundation_1
		}
	}
	if = {
		limit = {
			has_variable = ac_foundation_2
			var:ac_foundation_2 = scope:np_bad_tenet
		}
		if = {
			limit = { exists = scope:np_new_tenet }
			set_variable = { name = ac_foundation_2 value = scope:np_new_tenet }
		}
		else = {
			remove_variable = ac_foundation_2
		}
	}
}

# Rite scope, saved as scope:np_fix_rite. Add one allowed core tenet, if a valid one exists.
# First choice: a core tenet of the main rite that this rite lacks and can take.
# Fallback: a short fixed list. Every candidate must pass rite_can_choose_tenet (the game's own
# is_shown + can_pick check), so we never create an invalid tenet combination.
np_add_replacement_tenet_effect = {
	clear_saved_scope = np_new_tenet
	if = {
		limit = { this != faith.main_rite }
		faith.main_rite = {
			random_rite_tenet = {
				limit = {
					np_tenet_is_forbidden_trigger = no
					save_temporary_scope_as = np_cand_tenet
					scope:np_fix_rite = {
						NOT = { rite_has_tenet = scope:np_cand_tenet }
						rite_can_choose_tenet = scope:np_cand_tenet
					}
				}
				save_scope_as = np_new_tenet
			}
		}
	}
	# Second choice: any approved tenet the rite lacks and can take.
	if = {
		limit = { NOT = { exists = scope:np_new_tenet } }
		np_build_approved_tenet_list_effect = yes
		random_in_list = {
			list = np_approved_tenets
			limit = {
				save_temporary_scope_as = np_cand_tenet
				scope:np_fix_rite = {
					NOT = { rite_has_tenet = scope:np_cand_tenet }
					rite_can_choose_tenet = scope:np_cand_tenet
				}
			}
			save_scope_as = np_new_tenet
		}
	}
	if = {
		limit = { exists = scope:np_new_tenet }
		add_rite_tenet = scope:np_new_tenet
	}
	else = {
		debug_log = "Non Praevalebunt: no valid replacement tenet; the rite keeps one fewer core tenet"
		debug_log_scopes = yes
	}
}

# Inside pam_bull_events.0002, when np_bull_breaks_church_rules_trigger is true (see
# common/scripted_effects/zzz_np_bull_effects.txt). The Pope sets the bull aside: the tenet's status in the
# faith is restored, secret holders of a forbidden tenet are secret again, and the faithful get his letter.
np_set_aside_bull_effect = {
	scope:bull_issuer.faith = {
		if = {
			limit = { scope:bull_issuer.var:bull_option = flag:tenet_permitted }
			set_tenet_status = {
				tenet = scope:tenet_scope
				status = prohibited
			}
			pam_faith_grant_forbidden_personal_tenet_secrets_effect = { TENET = scope:tenet_scope }
		}
		else = {
			set_tenet_status = {
				tenet = scope:tenet_scope
				status = permitted
			}
			pam_faith_cleanup_forbidden_personal_tenet_secrets_effect = yes
		}
	}
	every_player = {
		limit = {
			np_character_in_church_trigger = yes
			this != scope:bull_issuer
		}
		trigger_event = { id = np_events.0002 days = 1 }
	}
	debug_log = "Non Praevalebunt: the Pope set aside a bull that broke the Church's rules"
}

# Inside ecumenical_council_conservative_backlash_effect (zzz_np_lock_effects.txt), which every council runs
# right before its tenet ruling takes effect (base game and Shadows of the Cross alike). Within the Church, a
# ruling that would Permit or make core a forbidden tenet, or Prohibit an approved one, is not confirmed by the
# Pope: the proposal becomes a flag the ruling ignores, so nothing changes, and the faithful get his letter.
# In the ruling's preview only the tooltip shows; nothing runs.
np_council_veto_tenet_ruling_effect = {
	if = {
		limit = { np_council_tenet_ruling_breaks_rules_trigger = yes }
		custom_tooltip = np_council_ruling_set_aside_tt
		hidden_effect = {
			scope:activity.var:current_topic = { save_scope_as = np_council_tenet }
			if = {
				limit = { scope:activity.var:current_debate_proposal ?= flag:prohibit }
				save_scope_value_as = { name = np_council_ruling value = flag:prohibit }
			}
			else = {
				save_scope_value_as = { name = np_council_ruling value = flag:permit }
			}
			scope:activity = {
				set_variable = { name = current_debate_proposal value = flag:np_set_aside }
			}
			np_council_send_letter_effect = yes
		}
		debug_log = "Non Praevalebunt: the Pope did not confirm a council's tenet ruling"
	}
}

# Inside ecumenical_council_host_rite_adopt_doctrine_effect (zzz_np_lock_effects.txt), which every council runs
# right after it changes one of the faith's doctrines. Within the Church, a forbidden doctrine is not confirmed:
# the faith goes back to the doctrine it held (the debate's topic), the guard checks the main rite, and the
# faithful get the Pope's letter.
np_council_revert_doctrine_effect = {
	custom_tooltip = np_council_ruling_set_aside_tt
	hidden_effect = {
		scope:ec_proposal_doctrine = { save_scope_as = np_council_doctrine }
		scope:host.faith = {
			if = {
				limit = { exists = scope:activity.var:current_topic }
				change_doctrine = scope:activity.var:current_topic
			}
			main_rite = { np_guard_rite_effect = yes }
		}
		np_council_send_letter_effect = yes
	}
	debug_log = "Non Praevalebunt: the Pope did not confirm a council's doctrine ruling"
}

# Players of the Church (except the Pope himself) get the Pope's letter about a council ruling he did not
# confirm (np_events.0003). Needs scope:np_council_tenet with scope:np_council_ruling, or scope:np_council_doctrine.
np_council_send_letter_effect = {
	every_player = {
		limit = {
			np_character_in_church_trigger = yes
			NOT = { has_title = title:k_papal_state }
		}
		trigger_event = { id = np_events.0003 days = 1 }
	}
}

# Rite scope. Rule 2: a rite in good standing never crosses the heresy line. When it comes within 5 points of
# it, the game moves it back toward the main rite (a negative divergence change, as in the base game's own
# backward drift), landing 10 points below the line. The game picks which allowed differences to drop.
np_pull_back_rite_effect = {
	if = {
		limit = { np_rite_needs_pull_back_trigger = yes }
		change_rite_divergence = {
			value = np_pull_back_divergence_value
			override_tenets_first = no
		}
		np_guard_rite_effect = yes
		debug_log = "Non Praevalebunt: a rite in good standing was pulled back from the heresy line"
		debug_log_scopes = yes
	}
}

##############################################
# Pope as head of faith before the schism (game rule np_pope_head_before_schism)
##############################################

# Any scope. Before the schism the undivided Church has no head of faith in By God Alone. With the game
# rule on, make the Papacy its head, so the main rite always follows the Pope (who is always Roman).
# Rome keeps the primacy in every case: if the schism is prevented and the peacemaker seats another see
# (e.g. Constantinople), or anything else gives the undivided Church another head, the Papacy is restored.
# (A peacemaker's "leave the seat vacant" changes nothing in the game, so the Pope simply stays head.)
np_pope_head_setup_effect = {
	if = {
		limit = {
			has_game_rule = np_pope_head_before_schism_on
			exists = rite:roman_rite
			rite:roman_rite.faith = faith:christian_faith
			NOT = { faith:christian_faith.religious_head_title ?= title:k_papal_state }
		}
		# Tell the undivided Church's players when another seat is being replaced.
		if = {
			limit = { exists = faith:christian_faith.religious_head_title }
			every_player = {
				limit = { faith = faith:christian_faith }
				send_interface_message = {
					type = event_generic_neutral
					title = np_msg_rome_keeps_primacy
					desc = np_msg_rome_keeps_primacy_desc
				}
			}
		}
		# Same order as By God Alone's own seating (pam_schism_seat_head_of_faith_effect): the faith must
		# have the Spiritual Head doctrine before it can be given a head-of-faith title.
		faith:christian_faith = {
			if = {
				limit = { NOT = { has_doctrine = doctrine_spiritual_head } }
				change_doctrine = doctrine_spiritual_head
			}
			set_religious_head_title = title:k_papal_state
			# Head-of-faith doctrines belong to the main rite only. Side rites keep theirs (a heresy
			# such as the Cathars may hold "No Head"); every rite just refreshes its head.
			main_rite = {
				if = {
					limit = { rite_has_doctrine = doctrine_no_head }
					change_rite_doctrine = doctrine_spiritual_head
				}
			}
			every_faith_rite = {
				force_refresh_head_of_rite = yes
			}
		}
		debug_log = "Non Praevalebunt: the Pope is now head of the undivided Church"
	}
}

# Any scope. After the schism the game makes the Papacy head of Catholicism. Make sure the empty old
# undivided faith no longer claims the same title.
np_schism_head_cleanup_effect = {
	if = {
		limit = {
			exists = rite:roman_rite
			NOT = { rite:roman_rite.faith = faith:christian_faith }
			faith:christian_faith.religious_head_title ?= title:k_papal_state
		}
		faith:christian_faith = { remove_religious_head_title = yes }
		debug_log = "Non Praevalebunt: removed the Papacy as head of the old undivided faith after the schism"
	}
}

# Be Fruitful and Multiply: blessed children
##############################################

# Root = the newborn (on_birth_child). If the child was born of a marriage, each parent whose own rite
# is in the Church and holds the np_blessed_children_piety parameter (Be Fruitful and Multiply, see
# np_tenet_overrides.txt) gains piety.
# "Born of a marriage": the mother and the family father are (or were, until death) married, so a
# posthumous child counts. Children of concubines are legitimate in the game but don't count.
np_blessed_children_effect = {
	if = {
		limit = {
			scope:is_bastard = no
			exists = scope:mother
			exists = scope:father
			scope:father = { is_spouse_of_even_if_dead = scope:mother }
		}
		scope:mother = { np_blessed_children_parent_effect = yes }
		scope:father = { np_blessed_children_parent_effect = yes }
	}
}

# Character scope (a parent).
np_blessed_children_parent_effect = {
	if = {
		limit = {
			is_alive = yes
			rite ?= {
				np_rite_in_church_trigger = yes
				rite_has_parameter = np_blessed_children_piety
			}
		}
		add_piety = minor_piety_gain
	}
}

# Root = the holder of title:k_papal_state (never the Shadows of the Cross steward; callers exclude him).
np_pope_take_roman_rite_effect = {
	# Every Pope belongs to the Church. Vanilla's Great Schism effect makes the same correction.
	if = {
		limit = { NOT = { faith = rite:roman_rite.faith } }
		set_character_faith = rite:roman_rite.faith
		debug_log = "Non Praevalebunt: the Pope was brought into the Church's faith"
		debug_log_scopes = yes
	}
	if = {
		limit = { NOT = { rite = rite:roman_rite } }
		set_character_rite = rite:roman_rite
		debug_log = "Non Praevalebunt: the Pope took the Roman rite"
		debug_log_scopes = yes
	}
	# When the Pope heads the faith (always after the schism; before it with the game rule on), the main
	# rite must be his Roman rite. Only when he really is the head: if a prevented schism seated another
	# head, that head's rite rules and we don't fight it.
	if = {
		limit = {
			faith.religious_head ?= this
			NOT = { faith.main_rite = rite:roman_rite }
		}
		rite:roman_rite = { set_main_rite = yes }
		debug_log = "Non Praevalebunt: the Roman rite is the main rite again"
	}
}

# Rite scope. An outbreak hit or created this rite: unless it is a historic rite (Tier 1, which never falls), it
# falls. It is no longer guarded: the guard and the pull-back leave it alone from now on.
np_rite_mark_fallen_effect = {
	if = {
		limit = {
			np_rite_in_church_trigger = yes
			np_rite_is_tier1_trigger = no
			NOT = { has_variable = np_fallen }
		}
		set_variable = np_fallen
		remove_variable = np_good_standing
		debug_log = "Non Praevalebunt: a rite fell into heresy"
		debug_log_scopes = yes
	}
}

# Rite scope. Mark the rite fallen (if it can fall) and give it its signature heresy.
np_rite_falls_effect = {
	np_rite_mark_fallen_effect = yes
	if = {
		limit = {
			has_variable = np_fallen
			np_rite_in_church_trigger = yes
			NOT = { has_variable = np_signature }
		}
		np_signature_heresy_effect = yes
	}
}

# Rite scope, a fallen rite of the Church. One heretical core tenet, the classic heresy of its place and era, so
# the game sees a real heresy: one clear error, as real heresies had (the Adoptionists, the Cathars...).
# If the rite already holds a forbidden tenet of its own, that one is its signature. Otherwise the rite is first
# brought back to about 30, then one tenet it does not share with the main rite (and that is not one of Shadows of
# the Cross's canon tenets) is swapped for the heresy, landing it at 75 to 90. The heresy is then marked Prohibited
# in the Church, and np_heresy_keep_past_line_effect corrects the landing if needed.
np_signature_heresy_effect = {
	clear_saved_scope = np_sig_out
	clear_saved_scope = np_sig_tenet
	save_scope_as = np_sig_rite
	if = {
		limit = { np_rite_has_forbidden_tenet_trigger = yes }
		random_rite_tenet = {
			limit = { np_tenet_is_forbidden_trigger = yes }
			save_scope_as = np_sig_tenet
		}
	}
	else = {
		if = {
			limit = { "divergence(faith.main_rite)" > 35 }
			change_rite_divergence = {
				value = np_signature_room_value
				override_tenets_first = no
			}
		}
		np_pick_signature_tenet_effect = yes
		faith.main_rite = {
			every_rite_tenet = { add_to_temporary_list = np_main_rite_tenets }
		}
		random_rite_tenet = {
			limit = {
				NOT = { is_in_list = np_main_rite_tenets }
				trigger_if = {
					limit = { scope:np_sig_rite = { has_variable = ac_foundation_1 } }
					NOT = { this = scope:np_sig_rite.var:ac_foundation_1 }
				}
				trigger_if = {
					limit = { scope:np_sig_rite = { has_variable = ac_foundation_2 } }
					NOT = { this = scope:np_sig_rite.var:ac_foundation_2 }
				}
			}
			save_scope_as = np_sig_out
		}
		# A rite that shares every tenet with the main rite gives up one of those instead (still never a canon tenet).
		if = {
			limit = { NOT = { exists = scope:np_sig_out } }
			random_rite_tenet = {
				limit = {
					trigger_if = {
						limit = { scope:np_sig_rite = { has_variable = ac_foundation_1 } }
						NOT = { this = scope:np_sig_rite.var:ac_foundation_1 }
					}
					trigger_if = {
						limit = { scope:np_sig_rite = { has_variable = ac_foundation_2 } }
						NOT = { this = scope:np_sig_rite.var:ac_foundation_2 }
					}
				}
				save_scope_as = np_sig_out
			}
		}
		if = {
			limit = { exists = scope:np_sig_out }
			remove_rite_tenet = scope:np_sig_out
		}
		add_rite_tenet = scope:np_sig_tenet
	}
	if = {
		limit = { exists = scope:np_sig_tenet }
		set_variable = { name = np_signature value = scope:np_sig_tenet }
	}
	faith = { np_mark_forbidden_tenets_effect = yes }
	np_heresy_keep_past_line_effect = yes
	debug_log = "Non Praevalebunt: a fallen rite took its signature heresy"
	debug_log_scopes = yes
}

# Rite scope (scope:np_sig_rite). Saves scope:np_sig_tenet: the classic heresy of the rite's place and era, read
# from its head's capital (or else its founder's):
#   Iberia before the year 1000: Adoptionism (Elipandus of Toledo).
#   Near the East (Anatolia, Armenia, the Levant, Mesopotamia, Egypt): Monophysitism.
#   The Balkans and Occitania: Gnosticism, the Bogomil and Cathar line.
#   Anywhere else, and in rich worldly sees: Simony.
np_pick_signature_tenet_effect = {
	clear_saved_scope = np_sig_place
	clear_saved_scope = np_sig_tenet
	head_of_rite ?= {
		capital_county ?= { title_province = { save_scope_as = np_sig_place } }
	}
	if = {
		limit = { NOT = { exists = scope:np_sig_place } }
		rite_founder ?= {
			capital_county ?= { title_province = { save_scope_as = np_sig_place } }
		}
	}
	if = {
		limit = {
			exists = scope:np_sig_place
			current_date < 1000.1.1
			scope:np_sig_place = { geographical_region = world_europe_west_iberia }
		}
		tenet:tenet_adoptionism = { save_scope_as = np_sig_tenet }
	}
	else_if = {
		limit = {
			exists = scope:np_sig_place
			scope:np_sig_place = {
				OR = {
					geographical_region = world_asia_minor
					geographical_region = world_middle_east
					geographical_region = world_africa_north_east
				}
			}
		}
		tenet:tenet_monophysitism = { save_scope_as = np_sig_tenet }
	}
	else_if = {
		limit = {
			exists = scope:np_sig_place
			scope:np_sig_place = {
				OR = {
					geographical_region = world_europe_south_east
					geographical_region = custom_occitan
				}
			}
		}
		tenet:tenet_gnosticism = { save_scope_as = np_sig_tenet }
	}
	else = {
		tenet:tenet_simony = { save_scope_as = np_sig_tenet }
	}
}

# Rite scope, a fallen rite still in error. The rule stays true: a heretical belief always means the rite is past
# the line. At or under the line it is pushed 15 points past it; at 95 or more it is brought back to about 85, short
# of the automatic split at 100, and its signature heresy is put back if that removed it.
np_heresy_keep_past_line_effect = {
	if = {
		limit = { "divergence(faith.main_rite)" <= faith.heresy_threshold }
		change_rite_divergence = {
			value = np_heresy_topup_value
			include_unknown = yes
		}
	}
	else_if = {
		limit = { "divergence(faith.main_rite)" >= 95 }
		change_rite_divergence = {
			value = np_heresy_trim_value
			override_tenets_first = no
		}
		if = {
			limit = { has_variable = np_signature }
			var:np_signature = { save_scope_as = np_keep_sig }
			if = {
				limit = {
					NOT = { any_rite_tenet = { this = scope:np_keep_sig } }
				}
				add_rite_tenet = scope:np_keep_sig
			}
		}
	}
}

# Rite scope, every rite of the Church, monthly. Upkeep for fallen rites:
#   no signature yet (a rite that fell under v0.2, or a dangerous rite the guard caught): give it one;
#   no error left at all: it comes home as a rite of the Church in good standing;
#   still in error: keep it past the line, and make sure Rome condemns it.
np_fallen_upkeep_effect = {
	if = {
		limit = {
			has_variable = np_fallen
			np_rite_in_church_trigger = yes
		}
		if = {
			limit = { NOT = { has_variable = np_signature } }
			np_signature_heresy_effect = yes
		}
		else_if = {
			limit = { np_rite_holds_an_error_trigger = no }
			np_rite_returns_effect = yes
		}
		else = {
			np_heresy_keep_past_line_effect = yes
			np_condemnation_clock_effect = yes
		}
	}
}

# Rite scope. A fallen rite that holds no error any more is a rite of the Church in good standing again. History
# has this too: Felix of Urgell recanted, and Durand of Huesca led Waldensians back to Rome in 1208.
np_rite_returns_effect = {
	remove_variable = np_fallen
	remove_variable = np_signature
	remove_variable = np_condemn_pending
	remove_variable = np_condemn_wait
	set_variable = np_good_standing
	debug_log = "Non Praevalebunt: a fallen rite laid down its errors and came home"
	debug_log_scopes = yes
}

# Rite scope, a fallen rite past the line (P1 A). If Rome has not condemned it about 5 years after it was first seen
# past the line, the Pope condemns it with the game's own bull (pam_bull_events.0003): same letters, same effects.
# Only for an AI Pope who heads the faith; a rite that can't be declared a heresy (pam_rite_can_be_declared_heresy_trigger)
# waits.
np_condemnation_clock_effect = {
	if = {
		limit = {
			"divergence(faith.main_rite)" > faith.heresy_threshold
			exists = faith.religious_head
			faith.religious_head = { is_ai = yes }
		}
		if = {
			limit = { NOT = { has_variable = np_condemn_pending } }
			set_variable = np_condemn_pending
			set_variable = {
				name = np_condemn_wait
				years = 5
			}
		}
		else_if = {
			limit = {
				NOT = { has_variable = np_condemn_wait }
				pam_rite_can_be_declared_heresy_trigger = yes
			}
			np_condemn_fallen_rite_effect = yes
		}
	}
}

# Rite scope. The Pope condemns this rite: the same scopes and variables the heresy bull decision sets, then the
# game's own bull event, which turns the rite into a faith of its own and writes to the faithful.
np_condemn_fallen_rite_effect = {
	save_scope_as = selected_item
	save_scope_as = heresy_rite
	remove_variable = np_condemn_pending
	remove_variable = np_condemn_wait
	faith.religious_head = {
		save_scope_as = bull_issuer
		set_variable = {
			name = bull_option
			value = flag:heresy
			years = 1
		}
		set_variable = {
			name = enact_a_bull_rite_heresy_cooldown
			years = 5
		}
		trigger_event = pam_bull_events.0003
	}
	debug_log = "Non Praevalebunt: the Pope condemned a fallen rite"
	debug_log_scopes = yes
}

# Faith scope, the Church. Every forbidden tenet that a rite of the Church holds as core is marked Prohibited in
# the Church (if it is not already), so it counts fully toward the heresy line (30 points, or 60 for the classic
# heresies, which carry a x2 weight). The classic heresies already start Prohibited.
np_mark_forbidden_tenets_effect = {
	save_scope_as = np_status_church
	every_faith_rite = {
		every_rite_tenet = {
			limit = { np_tenet_is_forbidden_trigger = yes }
			save_scope_as = np_status_tenet
			if = {
				limit = {
					NOT = {
						scope:np_status_church.main_rite = {
							has_tenet_status = {
								tenet = scope:np_status_tenet
								status = prohibited
							}
						}
					}
				}
				scope:np_status_church = {
					set_tenet_status = {
						tenet = scope:np_status_tenet
						status = prohibited
					}
				}
			}
		}
	}
}

# Tenet scope, with scope:np_restore_church.
np_restore_one_tenet_effect = {
	save_scope_as = np_restore_tenet
	if = {
		limit = {
			np_tenet_is_allowed_trigger = yes
			scope:np_restore_church.main_rite = {
				has_tenet_status = {
					tenet = scope:np_restore_tenet
					status = prohibited
				}
			}
		}
		scope:np_restore_church = {
			set_tenet_status = {
				tenet = scope:np_restore_tenet
				status = permitted
			}
			pam_faith_cleanup_forbidden_personal_tenet_secrets_effect = yes
		}
		debug_log = "Non Praevalebunt: an approved tenet that had been Prohibited was restored"
		debug_log_scopes = yes
	}
}

# Rite scope. A rite that has just come into the Church (Seek Communion, a mended Schism, or a rite left behind in
# the old undivided faith): a rite of the Church in good standing, set right by the guard and kept from the line.
np_welcome_rite_effect = {
	remove_variable = np_fallen
	remove_variable = np_signature
	remove_variable = np_condemn_pending
	remove_variable = np_condemn_wait
	remove_variable = c2_spread_remaining_converts
	remove_variable = c3_spread_remaining_converts
	set_variable = np_good_standing
	force_refresh_head_of_rite = yes
	np_guard_rite_effect = yes
	np_pull_back_rite_effect = yes
}

# Character scope: the head of a heresy whose Seek Communion the Pope accepted (pam_seek_communion_reconcile_faith_effect).
np_true_return_effect = {
	faith = { save_scope_as = np_returning_faith }
	rite:roman_rite.faith = { save_scope_as = np_returning_church }
	scope:np_returning_faith = {
		# Absolution, as in the base game, but both traits go.
		every_faith_character = {
			limit = {
				OR = {
					has_trait = heresiarch
					has_trait = excommunicated
				}
			}
			if = {
				limit = { has_trait = heresiarch }
				remove_trait = heresiarch
			}
			if = {
				limit = { has_trait = excommunicated }
				remove_trait = excommunicated
			}
		}
		# The heresy stops spreading.
		remove_variable = c3_spread_remaining_converts
		if = {
			limit = { exists = religious_head_title }
			remove_religious_head_title = yes
		}
		every_faith_rite = { add_to_list = np_returning_rites }
	}
	every_in_list = {
		list = np_returning_rites
		set_can_convert_to_rite = no
		set_parent_faith = {
			target = scope:np_returning_church
			main = no
			include_derived = no
		}
	}
	every_in_list = {
		list = np_returning_rites
		np_welcome_rite_effect = yes
	}
	every_player = {
		limit = {
			np_character_in_church_trigger = yes
			NOT = { has_title = title:k_papal_state }
		}
		trigger_event = { id = np_events.0004 days = 1 }
	}
	debug_log = "Non Praevalebunt: a heresy returned to the Church through Seek Communion"
}

# Faith scope, the Church, monthly. After the Schism, any rite of ours left behind in the old undivided faith (for
# example a heresy that returned before 1054, which the game's Schism code does not move) joins Catholicism.
np_gather_stranded_rites_effect = {
	if = {
		limit = {
			exists = faith:christian_faith
			NOT = { this = faith:christian_faith }
		}
		save_scope_as = np_gather_church
		faith:christian_faith = {
			every_faith_rite = {
				limit = {
					rite_counties >= 1
					OR = {
						has_variable = np_good_standing
						has_variable = np_fallen
					}
				}
				add_to_list = np_stranded_rites
			}
		}
		every_in_list = {
			list = np_stranded_rites
			set_parent_faith = {
				target = scope:np_gather_church
				main = no
				include_derived = no
			}
			if = {
				limit = { NOT = { has_variable = np_fallen } }
				np_welcome_rite_effect = yes
			}
		}
	}
}
