namespace = np_events

##############################################
# 0001: Upon This Rock
# A letter from the Pope to each player of the Church, one day after the game starts.
# Sent by np_startup_notice_effect (common/scripted_effects/np_effects.txt).
##############################################

np_events.0001 = {
	type = letter_event
	opening = {
		first_valid = {
			triggered_desc = {
				trigger = { is_female = yes }
				desc = np_events.0001.opening.daughter
			}
			desc = np_events.0001.opening.son
		}
	}
	desc = np_events.0001.desc
	sender = scope:np_pope

	trigger = {
		exists = title:k_papal_state.holder
		NOT = { has_title = title:k_papal_state }
		np_character_in_church_trigger = yes
	}

	immediate = {
		title:k_papal_state.holder = { save_scope_as = np_pope }
	}

	option = {
		name = np_events.0001.a
	}
}

##############################################
# 0002: A bull set aside
# A letter from the Pope to each player of the Church, when he sets aside a bull that would Permit a
# forbidden tenet or Prohibit an approved one. Sent by np_set_aside_bull_effect instead of the usual
# "bull enacted" letter. Scopes from pam_bull_events.0002: bull_issuer, tenet_scope.
##############################################

np_events.0002 = {
	type = letter_event
	opening = np_events.0002.opening
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:bull_issuer.var:bull_option ?= flag:tenet_forbidden }
				desc = np_events.0002.desc.prohibit
			}
			desc = np_events.0002.desc.permit
		}
	}
	sender = scope:bull_issuer

	trigger = {
		exists = scope:bull_issuer
		exists = scope:tenet_scope
	}

	option = {
		name = np_events.0002.a
	}
}

##############################################
# 0003: A council ruling not confirmed (v0.2)
# A letter from the Pope to each player of the Church, when a council voted something that breaks the Church's
# rules and he would not confirm it. Sent by np_council_send_letter_effect. Scopes: np_council_tenet with the
# scope value np_council_ruling (flag:permit or flag:prohibit), or np_council_doctrine.
##############################################

np_events.0003 = {
	type = letter_event
	opening = np_events.0002.opening
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:np_council_doctrine }
				desc = np_events.0003.desc.doctrine
			}
			triggered_desc = {
				trigger = { scope:np_council_ruling ?= flag:prohibit }
				desc = np_events.0003.desc.prohibit
			}
			desc = np_events.0003.desc.permit
		}
	}
	sender = scope:np_pope

	trigger = {
		exists = title:k_papal_state.holder
		OR = {
			exists = scope:np_council_tenet
			exists = scope:np_council_doctrine
		}
	}

	immediate = {
		title:k_papal_state.holder = { save_scope_as = np_pope }
	}

	option = {
		name = np_events.0003.a
	}
}

##############################################
# 0004: A heresy comes home (v0.3)
# A letter from the Pope to each player of the Church, when a heresy's head sought communion, was accepted, and its
# rites rejoined the Church, laying down their errors. Sent by np_true_return_effect.
# Scopes: np_returning_faith (the heresy, now an empty shell, whose name still reads as before).
##############################################

np_events.0004 = {
	type = letter_event
	opening = np_events.0002.opening
	desc = np_events.0004.desc
	sender = scope:np_pope

	trigger = {
		exists = title:k_papal_state.holder
		exists = scope:np_returning_faith
	}

	immediate = {
		title:k_papal_state.holder = { save_scope_as = np_pope }
	}

	option = {
		name = np_events.0004.a
	}
}
